    private void papel(String nome, String descricao) {

    private Usuario conta(String nome, String username, String senha, String papel) {

    private void demonstracao(Barbearia loja) throws IOException {
        try (var imagem = new ClassPathResource("demo/soldado.jpg").getInputStream()) {
            loja.logo = imagem.readAllBytes();
            loja.tipoLogo = "image/jpeg";
            loja.versaoLogo = 1;
        }
        var cliente = conta("Alex Demo", "cliente", "ClienteDemo@2026", "CLIENTE");
        var conta1 = conta("Rafa Demo", "barbeiro", "BarbeiroDemo@2026", "BARBEIRO");
        var conta2 = conta("Luiz Demo", "luiz", "LuizDemo@2026", "BARBEIRO");
        var corte = catalogo.servico(catalogo.gravar(null, new CatalogoService.EntradaServico(

        catalogo.gravar(null, new CatalogoService.EntradaServico(
                "Barba", "Modelagem e acabamento.", new BigDecimal("30.00"), 30, true, true));
        catalogo.gravar(null, new CatalogoService.EntradaServico(
                "Corte e barba", "Corte completo com modelagem da barba.", new BigDecimal("65.00"), 60, true, true));
        var rafa = catalogo.barbeiro(catalogo.gravar(null, new CatalogoService.EntradaBarbeiro(
                "Rafa Demo", "DegradÃª, corte clÃ¡ssico e finalizaÃ§Ã£o", conta1.getId(), true)).id());
        var luiz = catalogo.barbeiro(catalogo.gravar(null, new CatalogoService.EntradaBarbeiro(
                "Luiz Demo", "Barba desenhada e cortes com tesoura", conta2.getId(), true)).id());
        loja.premio = corte;
        loja.fidelidadeAtiva = true;
        for (var dia : DayOfWeek.values()) {
            expediente(loja, null, dia);
            for (var profissional : Set.of(rafa, luiz)) expediente(loja, profissional, dia);
        }
        var agora = Regra.agora().withSecond(0).withNano(0);
        for (int dias = 9; dias >= 1; dias--) {
            var a = reserva(cliente, rafa, corte, agora.toLocalDate().minusDays(dias).atTime(10, 0), Status.CONCLUIDO);
            fidelidade.concluir(loja, a, a.fim);
        }
        reserva(cliente, rafa, corte, agora.minusMinutes(45), Status.CONFIRMADO);
    }

    private void expediente(Barbearia loja, Barbeiro profissional, DayOfWeek dia) {
        var e = new Expediente();
        e.barbearia = loja;
        e.barbeiro = profissional;
        e.dia = dia;
        e.inicio = LocalTime.of(9, 0);
        e.fim = LocalTime.of(19, 0);
        expedientes.save(e);
    }

    private Agendamento reserva(Usuario cliente, Barbeiro barbeiro, Servico servico, LocalDateTime inicio, Status status) {
        var a = new Agendamento();
        a.cliente = cliente;
        a.barbeiro = barbeiro;
        a.servico = servico;
        a.nomeServico = servico.nome;
        a.elegivelNaReserva = true;
        a.preco = servico.preco;
        a.inicio = inicio;
        a.fim = inicio.plusMinutes(servico.duracaoMinutos);
        a.status = status;
        return agenda.save(a);
    }

package br.ueg.trindade.braully_web2_fullstack.controller;

import java.util.Map;
import jakarta.validation.Valid;
import jakarta.validation.constraints.*;
import org.springframework.http.ResponseEntity;
import org.springframework.security.authentication.BadCredentialsException;
import org.springframework.security.core.Authentication;
import org.springframework.web.bind.annotation.*;
import br.ueg.trindade.braully_web2_fullstack.security.Identidade;
import br.ueg.trindade.braully_web2_fullstack.service.*;

@RestController
@RequestMapping("/api")
@CrossOrigin(origins = {"http://localhost:5173", "http://127.0.0.1:5173"})
public class AuthController {
    private final AuthService auth;
    private final UsuarioService usuarios;

    public AuthController(AuthService auth, UsuarioService usuarios) {
        this.auth = auth;
        this.usuarios = usuarios;
    }

    @PostMapping("/login")
    public ResponseEntity<?> login(@Valid @RequestBody LoginRequest request) {
        try {
            return ResponseEntity.ok(auth.login(request.username(), request.senha()));
        } catch (BadCredentialsException e) {
            return ResponseEntity.status(401).body(Map.of("erro", "UsuÃ¡rio ou senha invÃ¡lidos."));
        }
    }

    @PostMapping("/cadastro")
    public UsuarioService.Resumo cadastro(@Valid @RequestBody UsuarioService.Entrada entrada) {
        return usuarios.criar(entrada);
    }

    @GetMapping("/perfil")
    public UsuarioService.Resumo perfil(Authentication auth) {
        return usuarios.buscarPorId(Identidade.id(auth));
    }

    @PutMapping("/perfil")
    public UsuarioService.Resumo perfil(@Valid @RequestBody UsuarioService.Perfil entrada, Authentication auth) {
        return usuarios.atualizarPerfil(Identidade.id(auth), entrada);
    }

    @PostMapping("/logout")
    public void sair(Authentication auth) {
        usuarios.revogarSessoes(Identidade.id(auth));
    }

    public record LoginRequest(@NotBlank @Size(max = 40) String username, @NotBlank @Size(max = 72) String senha) {}
}

package br.ueg.trindade.braully_web2_fullstack.security;

import java.io.IOException;
import org.springframework.security.authentication.UsernamePasswordAuthenticationToken;
import org.springframework.security.core.authority.SimpleGrantedAuthority;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;
import br.ueg.trindade.braully_web2_fullstack.repository.UsuarioRepository;
import io.jsonwebtoken.JwtException;
import jakarta.servlet.*;
import jakarta.servlet.http.*;

@Component
public class JwtFilter extends OncePerRequestFilter {
    private final JwtUtil jwt;
    private final UsuarioRepository usuarios;

    public JwtFilter(JwtUtil jwt, UsuarioRepository usuarios) {
        this.jwt = jwt;
        this.usuarios = usuarios;
    }

    @Override
    protected boolean shouldNotFilter(HttpServletRequest request) {
        return !request.getServletPath().startsWith("/api/");
    }

    @Override
    protected void doFilterInternal(HttpServletRequest req, HttpServletResponse res, FilterChain chain)
            throws ServletException, IOException {
        String header = req.getHeader("Authorization");
        if (header != null && header.startsWith("Bearer ")) {
            autenticar(header.substring(7));
        }
        chain.doFilter(req, res);
    }

    private void autenticar(String token) {
        try {
            var claims = jwt.extrairClaims(token);
            Long versao = claims.get("versao", Long.class);
            usuarios.findWithPermissoesById(Long.valueOf(claims.getSubject()))
                    .filter(u -> u.isAtivo() && versao != null && u.getVersaoSessao() == versao)
                    .ifPresent(usuario -> {
                        var papeis = usuario.getPermissoes().stream()
                                .map(p -> new SimpleGrantedAuthority("ROLE_" + p.getNome())).toList();
                        var identidade = new Identidade(usuario.getId(), usuario.getUsername());
                        SecurityContextHolder.getContext().setAuthentication(
                                new UsernamePasswordAuthenticationToken(identidade, null, papeis));
                    });
        } catch (JwtException | IllegalArgumentException ignored) {
            SecurityContextHolder.clearContext();
        }
    }
}

package br.ueg.trindade.braully_web2_fullstack.security;

import java.nio.charset.StandardCharsets;
import java.util.Date;
import java.util.List;

import javax.crypto.SecretKey;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;

import br.ueg.trindade.braully_web2_fullstack.model.Permissao;
import br.ueg.trindade.braully_web2_fullstack.model.Usuario;
import io.jsonwebtoken.Claims;
import io.jsonwebtoken.Jwts;
import io.jsonwebtoken.security.Keys;

@Component
public class JwtUtil {

    @Value("${jwt.validade-ms}")
    private long validadeMs;

    @Value("${jwt.secret}")
    private String secret;

    private SecretKey chave;

    @jakarta.annotation.PostConstruct
    void inicializarChave() {
        chave = secret.isBlank() ? Jwts.SIG.HS256.key().build()
                : Keys.hmacShaKeyFor(secret.getBytes(StandardCharsets.UTF_8));
    }

    public String gerarToken(Usuario usuario) {
        List<String> permissoes = usuario.getPermissoes().stream()
                .map(Permissao::getNome)
                .toList();

        return Jwts.builder()
                .subject(usuario.getId().toString())
                .claim("versao", usuario.getVersaoSessao())
                .id(java.util.UUID.randomUUID().toString())
                .claim("permissoes", permissoes)
                .issuedAt(new Date())
                .expiration(new Date(System.currentTimeMillis() + validadeMs))
                .signWith(chave)
                .compact();
    }

    public Claims extrairClaims(String token) {
        return Jwts.parser()
                .verifyWith(chave)
                .build()
                .parseSignedClaims(token)
                .getPayload();
    }
}

package br.ueg.trindade.braully_web2_fullstack.security;

import java.util.List;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.web.servlet.FilterRegistrationBean;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.http.HttpMethod;
import org.springframework.security.config.Customizer;
import org.springframework.security.config.annotation.method.configuration.EnableMethodSecurity;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.annotation.web.configuration.EnableWebSecurity;
import org.springframework.security.config.http.SessionCreationPolicy;
import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.security.web.SecurityFilterChain;
import org.springframework.security.web.authentication.UsernamePasswordAuthenticationFilter;
import org.springframework.web.cors.CorsConfiguration;
import org.springframework.web.cors.CorsConfigurationSource;
import org.springframework.web.cors.UrlBasedCorsConfigurationSource;

@Configuration
@EnableWebSecurity
@EnableMethodSecurity
public class SecurityConfig {

    @Autowired
    private JwtFilter jwtFilter;

    @Bean
    public PasswordEncoder passwordEncoder() {
        return new BCryptPasswordEncoder();
    }

    @Bean
    public SecurityFilterChain filterChain(HttpSecurity http) throws Exception {
        http
                .cors(Customizer.withDefaults())
                .csrf(csrf -> csrf.disable())
                .headers(headers -> headers
                        .frameOptions(frame -> frame.deny())
                        .contentSecurityPolicy(policy -> policy.policyDirectives(
                            "default-src 'self'; script-src 'self'; style-src 'self' 'unsafe-inline'; img-src 'self' blob:; frame-ancestors 'none'; base-uri 'self'; form-action 'self'")))
                .sessionManagement(sm -> sm.sessionCreationPolicy(SessionCreationPolicy.STATELESS))
                .exceptionHandling(ex -> ex
                        .authenticationEntryPoint((req, res, erro) -> responder(res, 401, "Entre na sua conta para continuar."))
                        .accessDeniedHandler((req, res, erro) -> responder(res, 403, "VocÃª nÃ£o tem permissÃ£o para esta operaÃ§Ã£o.")))
                .authorizeHttpRequests(auth -> auth
                        .requestMatchers(HttpMethod.OPTIONS, "/**").permitAll()
                        .requestMatchers(HttpMethod.GET, "/api/publico/**").permitAll()
                        .requestMatchers(HttpMethod.POST, "/api/login", "/api/cadastro").permitAll()
                        .requestMatchers("/api/**").authenticated()
                        .anyRequest().permitAll())
                .addFilterBefore(jwtFilter, UsernamePasswordAuthenticationFilter.class);

        return http.build();
    }

    private static void responder(jakarta.servlet.http.HttpServletResponse res, int status, String texto)
            throws java.io.IOException {
        res.setStatus(status);
        res.setContentType("application/json;charset=UTF-8");
        res.getWriter().write("{\"erro\":\"" + texto + "\"}");
    }

    // O JwtFilter Ã© um @Component: sem isto o Boot tambÃ©m o registraria no
    // encadeamento de servlets e ele rodaria duas vezes por requisiÃ§Ã£o.
    @Bean
    public FilterRegistrationBean<JwtFilter> jwtFilterRegistration(JwtFilter filter) {
        FilterRegistrationBean<JwtFilter> registro = new FilterRegistrationBean<>(filter);
        registro.setEnabled(false);
        return registro;
    }

    @Bean
    public CorsConfigurationSource corsConfigurationSource() {
        CorsConfiguration configuracao = new CorsConfiguration();
        configuracao.setAllowedOrigins(List.of("http://localhost:5173", "http://127.0.0.1:5173"));
        configuracao.setAllowedMethods(List.of("GET", "POST", "PUT", "DELETE", "OPTIONS"));
        configuracao.setAllowedHeaders(List.of("*"));

        UrlBasedCorsConfigurationSource origem = new UrlBasedCorsConfigurationSource();
        origem.registerCorsConfiguration("/**", configuracao);
        return origem;
    }

package br.ueg.trindade.braully_web2_fullstack.service;

import java.math.BigDecimal;
import java.time.*;
import java.util.List;
import jakarta.validation.constraints.NotNull;
import org.springframework.security.core.Authentication;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import br.ueg.trindade.braully_web2_fullstack.model.*;
import br.ueg.trindade.braully_web2_fullstack.model.Agendamento.Status;
import br.ueg.trindade.braully_web2_fullstack.repository.AgendamentoRepository;
import br.ueg.trindade.braully_web2_fullstack.security.Identidade;

@Service
@Transactional
public class AgendaService {
    private final AgendamentoRepository agenda;
    private final InstalacaoService instalacao;
    private final CatalogoService catalogo;
    private final UsuarioService usuarios;
    private final DisponibilidadeService disponibilidade;
    private final FidelidadeService fidelidade;

    public AgendaService(AgendamentoRepository agenda, InstalacaoService instalacao, CatalogoService catalogo,
                         UsuarioService usuarios, DisponibilidadeService disponibilidade, FidelidadeService fidelidade) {
        this.agenda = agenda;
        this.instalacao = instalacao;
        this.catalogo = catalogo;
        this.usuarios = usuarios;
        this.disponibilidade = disponibilidade;
        this.fidelidade = fidelidade;
    }

    public record Nova(@NotNull Long servicoId, Long barbeiroId, @NotNull LocalDateTime inicio, Long premioId,
                       Long clienteId) {}
    public record Reagendar(@NotNull Long barbeiroId, @NotNull LocalDateTime inicio) {}
    public record View(Long id, Long clienteId, String cliente, Long barbeiroId, String barbeiro, Long servicoId,
                       String servico, BigDecimal preco, LocalDateTime inicio, LocalDateTime fim, Status status,
                       boolean beneficio, boolean podeAlterar) {}

    @Transactional(readOnly = true)
    public List<View> listar(Authentication auth, LocalDate data, Long barbeiro) {
        if (!equipe(auth)) return listarCliente(Identidade.id(auth), false);
        LocalDate dia = data == null ? Regra.agora().toLocalDate() : data;
        return agenda.findByInicioGreaterThanEqualAndInicioLessThanOrderByInicioAsc(dia.atStartOfDay(), dia.plusDays(1).atStartOfDay())
                .stream().filter(a -> (barbeiro == null || a.barbeiro.getId().equals(barbeiro)) && atende(a, auth))
                .map(a -> view(a, true, 0)).toList();
    }

    @Transactional(readOnly = true)
    public List<View> proximosDaSemana() {
        var agora = Regra.agora();
        return agenda.findByInicioGreaterThanEqualAndInicioLessThanOrderByInicioAsc(agora, agora.plusDays(7))
                .stream().filter(DisponibilidadeService::pendente).limit(6).map(a -> view(a, true, 0)).toList();
    }

    @Transactional(readOnly = true)
    public List<View> listarCliente(Long clienteId, boolean equipe) {
        int antecedencia = instalacao.atual().antecedenciaMinutos;
        return agenda.findByClienteIdOrderByInicioDesc(clienteId).stream().map(a -> view(a, equipe, antecedencia)).toList();
    }

    @Transactional(readOnly = true)
    public View buscar(Long id, Authentication auth) {
        var a = autorizado(id, auth);
        return view(a, equipe(auth), instalacao.atual().antecedenciaMinutos);
    }

    @Transactional(readOnly = true)
    public List<LocalDateTime> horariosReagendamento(Long id, Long barbeiroId, LocalDate data, Authentication auth) {
        var a = autorizado(id, auth);
        alterar(a, auth, instalacao.atual());
        var b = destino(a, barbeiroId, auth);
        return b.ativo ? disponibilidade.livres(b, data, duracao(a), id) : List.of();
    }

    public View criar(Nova e, Authentication auth) {
        var loja = instalacao.bloquear();
        boolean admin = Identidade.tem(auth, "ADMIN");
        if (!admin && !Identidade.tem(auth, "CLIENTE")) throw Regra.proibida();
        if (e.clienteId() != null && !admin) throw Regra.proibida();
        if (admin && e.clienteId() == null) throw Regra.invalida("Selecione o cliente da reserva.");
        Long clienteId = e.clienteId() == null ? Identidade.id(auth) : e.clienteId();
        Usuario cliente = usuarios.buscar(clienteId);
        if (!cliente.isAtivo() || cliente.getPermissoes().stream().noneMatch(p -> p.getNome().equals("CLIENTE"))) {
            throw Regra.invalida("Selecione um cliente ativo.");
        }
        var s = catalogo.servico(e.servicoId());
        if (!s.ativo) throw Regra.conflito("Esse serviÃ§o estÃ¡ desativado.");
        validarAgendaDoCliente(clienteId, e.inicio(), s.duracaoMinutos, null);
        var b = e.barbeiroId() == null ? disponibilidade.atribuir(e.inicio(), s.duracaoMinutos) : catalogo.barbeiro(e.barbeiroId());
        validarHorario(b, e.inicio(), s.duracaoMinutos, null);
        var a = new Agendamento();
        a.barbeiro = b;
        a.cliente = cliente;
        a.servico = s;
        a.nomeServico = s.nome;
        a.beneficio = e.premioId() != null;
        a.elegivelNaReserva = s.elegivel;
        a.preco = a.beneficio ? BigDecimal.ZERO : s.preco;
        a.inicio = e.inicio();
        a.fim = e.inicio().plusMinutes(s.duracaoMinutos);
        agenda.save(a);
        if (a.beneficio) fidelidade.vincular(e.premioId(), a);
        return view(a, admin, loja.antecedenciaMinutos);
    }

    public View reagendar(Long id, Reagendar e, Authentication auth) {
        var loja = instalacao.bloquear();
        var a = autorizado(id, auth);
        alterar(a, auth, loja);
        var b = destino(a, e.barbeiroId(), auth);
        int duracao = duracao(a);
        validarAgendaDoCliente(a.cliente.getId(), e.inicio(), duracao, id);
        validarHorario(b, e.inicio(), duracao, a.id);
        a.barbeiro = b;
        a.inicio = e.inicio();
        a.fim = e.inicio().plusMinutes(duracao);
        a.status = Status.RESERVADO;
        return view(a, equipe(auth), loja.antecedenciaMinutos);
    }

    public View status(Long id, Status destino, Authentication auth) {
        var loja = instalacao.bloquear();
        var a = autorizado(id, auth);
        if (!equipe(auth) && destino != Status.CANCELADO) throw Regra.proibida();
        alterar(a, auth, loja);
        switch (destino) {
            case CONFIRMADO -> {
                if (a.status != Status.RESERVADO) throw Regra.conflito("Somente uma reserva pode ser confirmada.");
            }
            case CONCLUIDO -> {
                if (a.status != Status.CONFIRMADO) throw Regra.conflito("Confirme o atendimento antes de concluir.");
                if (a.fim.isAfter(Regra.agora())) throw Regra.conflito("Aguarde o fim do horÃ¡rio para concluir o atendimento.");
                fidelidade.concluir(loja, a);
            }
            case FALTOU -> {
                if (a.inicio.isAfter(Regra.agora())) throw Regra.conflito("Ainda nÃ£o Ã© possÃ­vel registrar a falta.");
                fidelidade.liberar(a);
            }
            case CANCELADO -> fidelidade.liberar(a);
            default -> throw Regra.conflito("TransiÃ§Ã£o de status invÃ¡lida.");
        }
        a.status = destino;
        return view(a, equipe(auth), loja.antecedenciaMinutos);
    }

    private void alterar(Agendamento a, Authentication auth, Barbearia loja) {
        if (!DisponibilidadeService.pendente(a)) throw Regra.conflito("Esse atendimento jÃ¡ foi encerrado.");
        if (!equipe(auth) && !a.inicio.isAfter(Regra.agora().plusMinutes(loja.antecedenciaMinutos))) {
            throw Regra.conflito("O prazo para cancelar ou reagendar pelo site terminou.");
        }
    }

    private int duracao(Agendamento a) { return Math.toIntExact(Duration.between(a.inicio, a.fim).toMinutes()); }

    private void validarAgendaDoCliente(Long clienteId, LocalDateTime inicio, int duracao, Long ignorar) {
        var fim = inicio.plusMinutes(duracao);
        if (agenda.findByClienteIdAndInicioLessThanAndFimGreaterThan(clienteId, fim, inicio).stream()
                .anyMatch(a -> !a.id.equals(ignorar) && DisponibilidadeService.pendente(a))) {
            throw Regra.conflito("JÃ¡ existe uma reserva para este cliente nesse intervalo.");
        }
    }

    private void validarHorario(Barbeiro b, LocalDateTime inicio, int duracao, Long ignorar) {
        if (!b.ativo || !disponibilidade.livres(b, inicio.toLocalDate(), duracao, ignorar).contains(inicio)) {
            throw Regra.conflito("Esse horÃ¡rio nÃ£o estÃ¡ disponÃ­vel. Escolha outro horÃ¡rio.");
        }
    }

    private Barbeiro destino(Agendamento a, Long id, Authentication auth) {
        if (Identidade.tem(auth, "BARBEIRO") && !Identidade.tem(auth, "ADMIN") && !a.barbeiro.getId().equals(id)) {
            throw Regra.proibida();
        }
        return catalogo.barbeiro(id);
    }

    private Agendamento autorizado(Long id, Authentication auth) {
        var a = agenda.findById(id).orElseThrow(Regra::inexistente);
        boolean autorizado = equipe(auth) ? atende(a, auth) : a.cliente.getId().equals(Identidade.id(auth));
        if (!autorizado) throw Regra.inexistente();
        return a;
    }

    private boolean equipe(Authentication auth) {
        return Identidade.tem(auth, "ADMIN") || Identidade.tem(auth, "BARBEIRO");
    }

    private boolean atende(Agendamento a, Authentication auth) {
        var conta = a.barbeiro.getUsuario();
        return Identidade.tem(auth, "ADMIN") || (conta != null && conta.getId().equals(Identidade.id(auth)));
    }

    private View view(Agendamento a, boolean equipe, int antecedencia) {
        boolean alterar = DisponibilidadeService.pendente(a)

    private record Intervalo(LocalDateTime inicio, LocalDateTime fim) {

        if (primeiroDia.isAfter(hoje.plusDays(90))) return List.of();

        for (var dia = primeiroDia; dia.isBefore(primeiroDia.plusDays(14))

    public List<LocalDateTime> livres(Barbeiro b, LocalDate dia, int duracao, Long ignorar) {
        LocalDateTime agora = Regra.agora();
        if (dia.isBefore(agora.toLocalDate()) || dia.isAfter(agora.toLocalDate().plusDays(90))) return List.of();
        var semana = expedientes.findByDia(dia.getDayOfWeek());
        var loja = semana.stream().filter(e -> e.barbeiro == null)
                .map(e -> new Intervalo(dia.atTime(e.inicio), dia.atTime(e.fim))).toList();
        var pessoal = semana.stream().filter(e -> doBarbeiro(e.barbeiro, b.id))
                .map(e -> new Intervalo(dia.atTime(e.inicio), dia.atTime(e.fim))).toList();
        if (loja.isEmpty() || pessoal.isEmpty()) return List.of();
        var inicioDia = dia.atStartOfDay();
        var fimDia = dia.plusDays(1).atStartOfDay();
        var ocupados = new ArrayList<Intervalo>();
        bloqueios.findByInicioLessThanAndFimGreaterThan(fimDia, inicioDia).stream()
                .filter(e -> e.barbeiro == null || doBarbeiro(e.barbeiro, b.id))
                .map(e -> new Intervalo(e.inicio, e.fim)).forEach(ocupados::add);
        agenda.findByBarbeiroIdAndInicioLessThanAndFimGreaterThan(b.id, fimDia, inicioDia).stream()
                .filter(a -> !a.id.equals(ignorar) && a.status != Agendamento.Status.CANCELADO && a.status != Agendamento.Status.FALTOU)
                .map(a -> new Intervalo(a.inicio, a.fim)).forEach(ocupados::add);
        var horarios = new ArrayList<LocalDateTime>();
        for (var hora = inicioDia; hora.isBefore(fimDia); hora = hora.plusMinutes(15)) {
            LocalDateTime inicio = hora;
            LocalDateTime fim = inicio.plusMinutes(duracao);
            if (inicio.isAfter(agora) && loja.stream().anyMatch(i -> i.inclui(inicio, fim))
                    && pessoal.stream().anyMatch(i -> i.inclui(inicio, fim))
                    && ocupados.stream().noneMatch(i -> i.sobrepoe(inicio, fim))) {
                horarios.add(inicio);
            }
        }
        return horarios;
    }

    private final MovimentoFidelidadeRepository movimentos;
    private final InstalacaoService instalacao;

    public FidelidadeService(MovimentoFidelidadeRepository movimentos, InstalacaoService instalacao) {
        this.movimentos = movimentos;

    }

    public record Movimento(Long id, Tipo tipo, int quantidade, String premio, Long servicoId,
                            LocalDateTime criadoEm, LocalDateTime resgatadoEm, Long agendamentoId) {}
    public record Cartao(boolean ativo, int progresso, int meta, String proximoPremio, List<Movimento> historico) {}

    @Transactional(readOnly = true)
    public Cartao cartao(Long clienteId) {
        Barbearia b = instalacao.atual();
        var lista = movimentos.findByClienteIdOrderByIdDesc(clienteId);
        return new Cartao(b.fidelidadeAtiva, saldo(lista), b.meta, b.premio == null ? null : b.premio.getNome(),
                lista.stream().map(m -> new Movimento(m.id, m.tipo, m.quantidade, m.nomePremio,
                        m.premio == null ? null : m.premio.getId(), m.criadoEm, m.resgatadoEm,
                        m.agendamento == null ? null : m.agendamento.getId())).toList());
    }

    private int saldo(List<MovimentoFidelidade> lista) {
        return lista.stream().mapToInt(m -> m.tipo == Tipo.CREDITO ? m.quantidade : -m.quantidade).sum();
    }

    // Chamado na transaÃ§Ã£o da agenda, com a linha da instalaÃ§Ã£o bloqueada.
    public void concluir(Barbearia b, Agendamento a) {
        concluir(b, a, Regra.agora());
    }

    public void concluir(Barbearia b, Agendamento a, LocalDateTime momento) {
        if (a.beneficio) {
            var premio = movimentos.findByAgendamentoIdAndTipo(a.id, Tipo.RECOMPENSA)
                    .orElseThrow(() -> Regra.conflito("O benefÃ­cio nÃ£o estÃ¡ vinculado Ã  reserva."));
            premio.resgatadoEm = momento;
            return;
        }
        if (!b.fidelidadeAtiva || !a.elegivelNaReserva || movimentos.existsByAgendamentoIdAndTipo(a.id, Tipo.CREDITO)) {
            return;
        }
        var credito = novo(a.cliente, Tipo.CREDITO, 1, momento);
        credito.agendamento = a;
        movimentos.save(credito);
        int saldo = saldo(movimentos.findByClienteIdOrderByIdDesc(a.cliente.getId()));
        while (saldo >= b.meta) {
            var premio = novo(a.cliente, Tipo.RECOMPENSA, b.meta, momento);
            premio.premio = b.premio;
            premio.nomePremio = b.premio.getNome();
            movimentos.save(premio);
            saldo -= b.meta;
        }
    }

    public void vincular(Long premioId, Agendamento a) {
        var m = movimentos.findById(premioId).orElseThrow(Regra::inexistente);
        if (m.tipo != Tipo.RECOMPENSA || !m.cliente.getId().equals(a.cliente.getId())) throw Regra.proibida();
        if (m.resgatadoEm != null || m.agendamento != null || !m.premio.getId().equals(a.servico.getId())) {
            throw Regra.conflito("O benefÃ­cio jÃ¡ foi reservado, utilizado ou pertence a outro serviÃ§o.");
        }
        m.agendamento = a;
    }

        if (!a.beneficio) return;
        movimentos.findByAgendamentoIdAndTipo(a.id, Tipo.RECOMPENSA).ifPresent(m -> m.agendamento = null);
    }

import br.ueg.trindade.braully_web2_fullstack.model.Foto;

import java.io.IOException;
import java.util.List;
import java.util.Objects;
import java.util.UUID;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.web.multipart.MultipartFile;

@Service
@Transactional
public class FotoService {
    private final FotoRepository fotos;
    private final InstalacaoService instalacao;
    private final CatalogoService catalogo;

    public FotoService(FotoRepository fotos, InstalacaoService instalacao, CatalogoService catalogo) {
        this.fotos = fotos;
        this.instalacao = instalacao;
        this.catalogo = catalogo;
    }

    public record FotoView(UUID id, Foto.Alvo alvo, Long cadastroId, String legenda, String categoria, String url) {}

    private FotoView view(UUID id, Foto.Alvo alvo, Long cadastroId, String legenda, String categoria) {
        return new FotoView(id, alvo, cadastroId, legenda, categoria, "/api/publico/fotos/" + id);
    }

    @Transactional(readOnly = true)
    public List<FotoView> listar() {
        return fotos.metadados(instalacao.atual().id).stream()
            .map(f -> view(f.getId(), f.getAlvo(), f.getCadastroId(), f.getLegenda(), f.getCategoria())).toList();
    }

    @Transactional(readOnly = true)
    public MarcaService.Imagem buscar(UUID id) {
        Foto foto = fotos.findById(id).orElseThrow(Regra::inexistente);
        return new MarcaService.Imagem(foto.conteudo, foto.tipo);
    }

    public FotoView salvar(Foto.Alvo alvo, Long cadastro, String legenda, String categoria, MultipartFile arquivo) throws IOException {
        if ((alvo == Foto.Alvo.SERVICO || alvo == Foto.Alvo.BARBEIRO) != (cadastro != null))
            throw Regra.invalida("Selecione um cadastro vÃ¡lido para a foto.");
        if (legenda != null && legenda.length() > 120) throw Regra.invalida("Use uma legenda de atÃ© 120 caracteres.");
        var imagem = MarcaService.validarImagem(arquivo);
        var loja = instalacao.bloquear();
        if (alvo == Foto.Alvo.SERVICO) catalogo.servico(cadastro);
        if (alvo == Foto.Alvo.BARBEIRO) catalogo.barbeiro(cadastro);
        var existentes = fotos.metadados(loja.id).stream()
            .filter(f -> f.getAlvo() == alvo && Objects.equals(f.getCadastroId(), cadastro)).toList();
        if (alvo == Foto.Alvo.GALERIA && existentes.size() >= 12)
            throw Regra.invalida("A galeria comporta atÃ© 12 fotos.");
        if (alvo != Foto.Alvo.GALERIA) existentes.forEach(f -> fotos.deleteById(f.getId()));
        var foto = new Foto();
        foto.barbearia = loja;
        foto.alvo = alvo;
        foto.cadastroId = cadastro;
        foto.legenda = legenda;
        if (alvo == Foto.Alvo.GALERIA) {
            if (categoria != null && !java.util.Set.of("Cortes", "Barba", "Ambiente", "Outros").contains(categoria))
                throw Regra.invalida("Selecione uma categoria de galeria vÃ¡lida.");
            foto.categoria = categoria == null ? "Outros" : categoria;
        }
        foto.tipo = imagem.tipo();
        foto.conteudo = imagem.bytes();
        fotos.save(foto);
        return view(foto.id, foto.alvo, foto.cadastroId, foto.legenda, foto.categoria);
    }

    public void remover(UUID id) {
        var loja = instalacao.bloquear();
        if (fotos.metadados(loja.id).stream().noneMatch(f -> f.getId().equals(id))) throw Regra.inexistente();
        fotos.deleteById(id);
    }
}

        return new Marca(b.nome, b.descricao, b.endereco, b.telefone, b.corDestaque, b.corFundo,
                imagem, b.demonstracao, b.fidelidadeAtiva, b.meta,

    }

        if (b.logo == null) throw Regra.inexistente();
        return new Imagem(b.logo, b.tipoLogo);
    }

    public Marca atualizar(Entrada e) {
        Barbearia b = instalacao.bloquear();
        Servico premio = e.premioId() == null ? null : catalogo.servico(e.premioId());
        if (e.fidelidadeAtiva() && (premio == null || !premio.ativo)) {
            throw Regra.invalida("Escolha um serviÃ§o ativo como recompensa.");
        }
        b.nome = e.nome().strip();
        b.descricao = e.descricao();
        b.endereco = e.endereco();
        b.telefone = e.telefone();
        if (e.instagram() != null) b.instagram = e.instagram();
        if (e.whatsapp() != null) b.whatsapp = e.whatsapp();
        b.corDestaque = e.corDestaque();
        b.corFundo = e.corFundo();
        b.demonstracao = e.demonstracao();
        b.fidelidadeAtiva = e.fidelidadeAtiva();

    public Marca enviarLogo(MultipartFile arquivo) throws IOException {
        Imagem imagem = validarImagem(arquivo);
        Barbearia b = instalacao.bloquear();
        b.logo = imagem.bytes();
        b.tipoLogo = imagem.tipo();
        b.versaoLogo++;
        return view(b);
    }

    static Imagem validarImagem(MultipartFile arquivo) throws IOException {
        if (arquivo.isEmpty() || arquivo.getSize() > 2_097_152) throw Regra.invalida("Selecione uma imagem de atÃ© 2 MB.");
        try (var entrada = arquivo.getInputStream(); var imagem = ImageIO.createImageInputStream(entrada)) {
            var leitores = ImageIO.getImageReaders(imagem);
            if (!leitores.hasNext()) throw Regra.invalida("Envie um PNG ou JPEG vÃ¡lido.");
            var leitor = leitores.next();
            try {
                leitor.setInput(imagem);
                String formato = leitor.getFormatName();
                if (!(formato.equalsIgnoreCase("JPEG") || formato.equalsIgnoreCase("PNG"))
                        || leitor.getWidth(0) > 5000 || leitor.getHeight(0) > 5000) {
                    throw Regra.invalida("Use um PNG ou JPEG com atÃ© 5000 pixels por lado.");
                }
                // Decodifica e reencoda para armazenar somente a imagem validada.
                var resultado = new java.io.ByteArrayOutputStream();
                ImageIO.write(leitor.read(0), formato, resultado);
                if (resultado.size() == 0 || resultado.size() > 2_097_152) throw Regra.invalida("A imagem processada excede 2 MB.");
                return new Imagem(resultado.toByteArray(), formato.equalsIgnoreCase("PNG") ? "image/png" : "image/jpeg");
            } finally {
                leitor.dispose();
            }
        } catch (javax.imageio.IIOException ex) {
            throw Regra.invalida("A imagem estÃ¡ danificada ou incompleta.");
        }
    }

    public Marca removerLogo() {
        Barbearia b = instalacao.bloquear();
        b.logo = null;
        b.tipoLogo = null;
        b.versaoLogo++;
        return view(b);
    }
}

package br.ueg.trindade.braully_web2_fullstack.service;

import java.util.*;
import java.nio.charset.StandardCharsets;
import jakarta.validation.constraints.*;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import br.ueg.trindade.braully_web2_fullstack.model.*;
import br.ueg.trindade.braully_web2_fullstack.repository.*;

@Service
@Transactional
public class UsuarioService {
    private final UsuarioRepository usuarios;
    private final PermissaoRepository permissoes;
    private final PasswordEncoder encoder;
    private final InstalacaoService instalacao;
    private final BarbeiroRepository barbeiros;

    public UsuarioService(UsuarioRepository usuarios, PermissaoRepository permissoes,
                          PasswordEncoder encoder, InstalacaoService instalacao, BarbeiroRepository barbeiros) {
        this.usuarios = usuarios;
        this.permissoes = permissoes;
        this.encoder = encoder;
        this.instalacao = instalacao;
        this.barbeiros = barbeiros;
    }

    public record Entrada(@NotBlank @Size(max = 100) String nome,
                          @NotBlank @Pattern(regexp = "[a-zA-Z0-9._-]{3,40}") String username,
                          @NotBlank @Email @Size(max = 150) String email, @Size(max = 72) String senha) {}
    public record Perfil(@NotBlank @Size(max = 100) String nome, @NotBlank @Email @Size(max = 150) String email,
                         @Size(max = 72) String senhaAtual, @Size(max = 72) String novaSenha) {}
    public record Resumo(Long id, String nome, String username, String email, boolean ativo, List<Permissao> permissoes) {}

    public static Resumo resumo(Usuario u) {
        return new Resumo(u.getId(), u.getNome(), u.getUsername(), u.getEmail(), u.isAtivo(),
                u.getPermissoes().stream().sorted(Comparator.comparing(Permissao::getNome)).toList());
    }

    @Transactional(readOnly = true)
    public List<Resumo> listarTodos() {
        return usuarios.findAll().stream().map(UsuarioService::resumo).sorted(Comparator.comparing(Resumo::nome)).toList();
    }

    @Transactional(readOnly = true)
    public Resumo buscarPorId(Long id) {
        return resumo(buscar(id));
    }

    public Usuario buscar(Long id) {
        return usuarios.findWithPermissoesById(id).orElseThrow(Regra::inexistente);
    }

    public Resumo criar(Entrada entrada) {
        return resumo(criarConta(entrada, "CLIENTE"));
    }

    public Usuario criarConta(Entrada e, String papel) {
        String username = e.username().toLowerCase(Locale.ROOT);
        Usuario u = new Usuario(e.nome().strip(), username, codificar(e.senha()), e.email().strip());
        u.getPermissoes().add(permissoes.findByNome(papel).orElseThrow());
        return usuarios.save(u);
    }

    public Resumo atualizar(Long id, Entrada e) {
        instalacao.bloquear();
        Usuario u = buscar(id);
        u.setNome(e.nome().strip());
        String username = e.username().toLowerCase(Locale.ROOT);
        if (!username.equals(u.getUsername())) {
            u.setUsername(username);
            u.revogarSessoes();
        }
        u.setEmail(e.email().strip());
        if (e.senha() != null && !e.senha().isBlank()) {
            u.setSenha(codificar(e.senha()));
            u.revogarSessoes();
        }
        return resumo(u);
    }

    public Resumo atualizarPerfil(Long id, Perfil e) {
        instalacao.bloquear();
        Usuario u = buscar(id);
        u.setNome(e.nome().strip());
        u.setEmail(e.email().strip());
        if (e.novaSenha() != null && !e.novaSenha().isBlank()) {
            if (e.senhaAtual() == null || e.senhaAtual().getBytes(StandardCharsets.UTF_8).length > 72
                    || !encoder.matches(e.senhaAtual(), u.getSenha())) {
                throw Regra.invalida("A senha atual estÃ¡ incorreta.");
            }
            u.setSenha(codificar(e.novaSenha()));
            u.revogarSessoes();
        }
        return resumo(u);
    }

    public void revogarSessoes(Long id) {
        instalacao.bloquear();
        buscar(id).revogarSessoes();
    }

    public void excluir(Long id, Long autor) {
        instalacao.bloquear();
        if (id.equals(autor)) throw Regra.conflito("VocÃª nÃ£o pode desativar a prÃ³pria conta.");
        Usuario usuario = buscar(id);
        usuario.setAtivo(false);
        usuario.revogarSessoes();
    }

    public Resumo ativar(Long id) {
        instalacao.bloquear();
        Usuario u = buscar(id);
        u.setAtivo(true);
        return resumo(u);
    }

    public Resumo atribuirPermissoes(Long id, List<Long> ids, Long autor) {
        instalacao.bloquear();
        if (id.equals(autor)) throw Regra.conflito("Sua prÃ³pria permissÃ£o nÃ£o pode ser alterada.");
        Usuario usuario = buscar(id);
        if (ids.isEmpty() || ids.contains(null)) throw Regra.invalida("Selecione ao menos uma permissÃ£o.");
        Set<Permissao> selecao = new HashSet<>(permissoes.findAllById(ids));
        if (selecao.size() != new HashSet<>(ids).size()) throw Regra.invalida("PermissÃ£o invÃ¡lida.");
        if (barbeiros.existsByUsuarioId(id) && selecao.stream().noneMatch(p -> p.getNome().equals("BARBEIRO"))) {
            throw Regra.conflito("Desvincule a conta do profissional antes de remover o perfil BARBEIRO.");
        }
        usuario.setPermissoes(selecao);
        usuario.revogarSessoes();
        return resumo(usuario);
    }

    private String codificar(String senha) {
        if (senha == null || senha.isBlank() || senha.length() < 8 || senha.getBytes(StandardCharsets.UTF_8).length > 72) {
            throw Regra.invalida("A senha precisa ter pelo menos 8 caracteres e no mÃ¡ximo 72 bytes.");
        }
        return encoder.encode(senha);
    }
}
