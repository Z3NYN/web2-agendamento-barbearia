import { ServicosPage, ProfissionaisPage, ExpedientePage } from "./pages/CatalogoPages";
import { ClientesPage, ClientePage } from "./pages/ClientesPage";
import MarcaPage from "./pages/MarcaPage";
import { obterToken, temPapel } from "./services/auth";

function RequireAuth({ papeis }: { papeis?: string[] }) {
  const { pathname, search } = useLocation();
  if (!obterToken()) return <Navigate to="/login" state={{ destino: pathname + search }} replace />;
  if (papeis && !papeis.some(temPapel)) return <Titulo titulo="Acesso restrito" detalhe="Seu perfil nÃ£o tem acesso a esta pÃ¡gina." />;
  return <Outlet />;
}

export default function App() {

    <Route path="login" element={<LoginPage />} />
    <Route path="cadastro" element={<LoginPage cadastro />} />
    <Route path="agendar" element={<AgendarPage />} />
    <Route element={<RequireAuth papeis={["CLIENTE"]} />}>
      <Route path="meus-horarios" element={<AgendaPage />} />
      <Route path="fidelidade" element={<FidelidadePage />} />

    <Route element={<RequireAuth />}>
      <Route path="perfil" element={<PerfilPage />} />

      <Route path="agenda" element={<AgendaPage equipe />} />

      <Route path="painel" element={<PainelPage />} />
      <Route path="servicos" element={<ServicosPage />} />

      <Route path="expediente" element={<ExpedientePage />} />

export function Vitrine({ titulo, children }: { titulo: string; children: ReactNode }) {
  const lista = useRef<HTMLDivElement>(null);
  const [pos, setPos] = useState(0);
  function mover(delta: number) {
    if (lista.current) lista.current.scrollBy({ left: lista.current.clientWidth * delta * .78 });
  }
  return <div className="vitrine">
    <div className="vitrine-track" ref={lista} tabIndex={0} role="region" aria-label={titulo}
      onScroll={e => {
        const t = e.currentTarget;
        setPos(t.scrollLeft / Math.max(1, t.scrollWidth - t.clientWidth));
      }}
      onKeyDown={e => {
        if (e.target !== e.currentTarget) return;
        if (e.key === "ArrowLeft" || e.key === "ArrowRight") {
          e.preventDefault(); mover(e.key === "ArrowLeft" ? -1 : 1);
        }
      }}>{children}</div>
    <div className="vitrine-controls">
      <span aria-hidden="true"><i style={{ transform: `translateX(${pos * 200}%)` }} /></span>
      <button type="button" className="icon-btn" onClick={() => mover(-1)} aria-label={`Voltar em ${titulo}`}><ChevronLeft size={18} /></button>
      <button type="button" className="icon-btn" onClick={() => mover(1)} aria-label={`AvanÃ§ar em ${titulo}`}><ChevronRight size={18} /></button>
    </div>
  </div>;
}

// src/services/auth.ts
const TOKEN = "token";
const USUARIO = "usuario";

export interface Sessao {
  token: string;
  nome: string;
  permissoes: string[];
}

export function salvarSessao(sessao: Sessao) {
  sessionStorage.setItem(TOKEN, sessao.token);
  sessionStorage.setItem(USUARIO, JSON.stringify(sessao));
}

export function obterToken(): string | null {
  return sessionStorage.getItem(TOKEN);
}

export function obterSessao(): Sessao | null {
  const usuario = sessionStorage.getItem(USUARIO);
  if (!usuario) return null;

  try {
    return JSON.parse(usuario) as Sessao;
  } catch {
    return null;
  }
}

export function temPapel(papel: string): boolean {
  const sessao = obterSessao();
  return sessao ? sessao.permissoes.includes(papel) : false;
}

export function sair() {
  sessionStorage.removeItem(TOKEN);
  sessionStorage.removeItem(USUARIO);
}

import { useCallback, useEffect, useState } from "react";
import api from "./api";
import { mensagem } from "./formato";

export function useConsulta<T>(url: string | null, intervalo = 0) {
  const [resultado, setResultado] = useState<{ url: string; dados?: T; erro?: string; desatualizado?: boolean }>();
  const [versao, setVersao] = useState(0);
  const recarregar = useCallback(() => {
    if (!intervalo) setResultado(undefined);
    setVersao(v => v + 1);
  }, [intervalo]);
  useEffect(() => {
    if (!url) return;
    const endereco = url;
    const controle = new AbortController();
    let timer: number | undefined;
    let consultando = false;
    async function consultar() {
      if (consultando || controle.signal.aborted) return;
      window.clearTimeout(timer);
      consultando = true;
      try {
        const r = await api.get<T>(endereco, { signal: controle.signal, timeout: intervalo ? 10000 : undefined });
        if (!controle.signal.aborted) setResultado(anterior => intervalo && anterior?.url === endereco && anterior.dados !== undefined && JSON.stringify(anterior.dados) === JSON.stringify(r.data)
          ? (anterior.desatualizado ? { ...anterior, desatualizado: false } : anterior) : { url: endereco, dados: r.data });
      } catch (e) {
        if (!controle.signal.aborted) setResultado(anterior => intervalo && anterior?.url === endereco && anterior.dados !== undefined
          ? { ...anterior, desatualizado: true } : { url: endereco, erro: mensagem(e) });
      } finally {
        consultando = false;
        if (intervalo && !controle.signal.aborted && !document.hidden) timer = window.setTimeout(reativar, intervalo);
      }
    }
    function reativar() { if (!document.hidden) void consultar(); }
    void consultar();
    if (intervalo) {
      document.addEventListener("visibilitychange", reativar);
      window.addEventListener("online", reativar);
    }
    return () => { controle.abort(); window.clearTimeout(timer); document.removeEventListener("visibilitychange", reativar); window.removeEventListener("online", reativar); };
  }, [url, versao, intervalo]);
  const atual = url && resultado?.url === url ? resultado : undefined;
  return { dados: atual?.dados, erro: atual?.erro, desatualizado: Boolean(atual?.desatualizado), carregando: Boolean(url && !atual), recarregar };
}
